獨立客戶端 Ed25519 簽章驗證測試器 (Verifier)

第三方使用者或微服務僅需持有系統公鑰（32 Bytes）、原始 Payload 與 64 Bytes 簽章，即可在完全無私鑰的情況下離線驗證真偽與資料完整性。

1. 固定私鑰加密保管 (HSM / KMS)

採用 AWS CloudHSM、GCP Cloud KMS 或 HashiCorp Vault 的非對稱簽署引擎。私鑰在硬體模組內生成且**不可匯出 (Non-exportable)**。微服務若需落盤持久化，強制使用信封加密 (Envelope Encryption) 或 PKCS#8 EncryptedPrivateKeyInfo (PBES2 / AES-256-GCM)，杜絕明文外洩。

  • 符合 FIPS 140-2 Level 3 認證標準
  • 靜態存儲強加密 (AES-256-GCM + PBKDF2 100k 迭代)
  • 內部記憶體抹除機制 (Zeroize on exit / Lock)
  • IAM 嚴格白名單與 RBAC 最小特權

2. 防重放與時效校驗

單純簽署 raw data 容易遭受重放攻擊 (Replay Attack)。生產級架構強制在簽名請求中包裹 Nonce、Timestamp (±30s) 與 Caller ID。

  • Redis 分散式 Nonce 鍵值快取 (TTL 5分鐘)
  • 防止惡意中間人截獲合規封包重複提交
  • 可指定 Context Domain Separation

3. 簽名審計日誌 (Audit Trail)

所有簽名操作皆不可抵賴 (Non-repudiation)。日誌系統同步記錄簽署請求雜湊、調用方身分、IP、簽名雜湊與授權 Token，寫入不可篡改的 WORM (Write-Once-Read-Many) 儲存庫。

  • 日誌不儲存未脫敏機敏 Payload
  • 即時 Prometheus 指標監控簽名頻率
  • 異常大量呼叫自動觸發熔斷告警

import crypto from 'node:crypto';

// 伺服器端簽署實作範例 (固定 Ed25519 私鑰)
export function signPayloadEd25519(payloadString, privateKeyDerBuffer) {
  const privateKey = crypto.createPrivateKey({
    key: privateKeyDerBuffer,
    format: 'der',
    type: 'pkcs8'
  });

  const dataBuffer = Buffer.from(payloadString, 'utf-8');
  
  // Ed25519 原生產出恰好 64 位元組 (512 bits) 的簽名值
  const signatureBuffer = crypto.sign(null, dataBuffer, privateKey);
  
  console.assert(signatureBuffer.length === 64, "Signature must be exactly 64 bytes!");

  return {
    signatureHex: signatureBuffer.toString('hex'),
    signatureBase64: signatureBuffer.toString('base64'),
    byteLength: signatureBuffer.length
  };
}

// 客戶端驗證實作範例 (持有 32-byte 公鑰)
export function verifySignature(payloadString, signatureHex, publicKeyRawBuffer) {
  const publicKey = crypto.createPublicKey({
    key: Buffer.concat([
      // Ed25519 SPKI 前置 ASN.1 標頭 (12 bytes) + 32 bytes raw key
      Buffer.from('302a300506032b6570032100', 'hex'),
      publicKeyRawBuffer
    ]),
    format: 'der',
    type: 'spki'
  });

  return crypto.verify(
    null,
    Buffer.from(payloadString, 'utf-8'),
    publicKey,
    Buffer.from(signatureHex, 'hex')
  );
}

from cryptography.hazmat.primitives.asymmetric import ed25519

# 1. 內部固定私鑰載入 (32-byte raw seed)
# 實務中由環境變數或 KMS 注入
PRIVATE_KEY_BYTES = bytes.fromhex("YOUR_32_BYTE_HEX_SEED")
private_key = ed25519.Ed25519PrivateKey.from_private_bytes(PRIVATE_KEY_BYTES)
public_key = private_key.public_key()

def sign_data(data_str: str) -> dict:
    data_bytes = data_str.encode('utf-8')
    
    # 產出標準 64 bytes Ed25519 Raw Signature
    signature = private_key.sign(data_bytes)
    assert len(signature) == 64, "簽名長度必須為 64 位元組"
    
    pub_bytes = public_key.public_bytes_raw()
    return {
        "signature_hex": signature.hex(),
        "signature_bytes_len": len(signature),
        "public_key_hex": pub_bytes.hex()
    }

# 2. 客戶端無私鑰驗證
def verify_data(data_str: str, signature_hex: str, public_key_hex: str) -> bool:
    try:
        verifier_key = ed25519.Ed25519PublicKey.from_public_bytes(bytes.fromhex(public_key_hex))
        verifier_key.verify(bytes.fromhex(signature_hex), data_str.encode('utf-8'))
        return True
    except Exception:
        return False
package main

import (
	"crypto/ed25519"
	"encoding/hex"
	"fmt"
)

// 內部固定金鑰對 (64-byte private key = 32-byte seed + 32-byte pubkey)
var internalPrivateKey ed25519.PrivateKey

// 從 32-byte hex seed 初始化內部固定金鑰
func InitKeyFromSeed(seedHex string) error {
	seedBytes, err := hex.DecodeString(seedHex)
	if err != nil || len(seedBytes) != 32 {
		return fmt.Errorf("seed 必須為 32 位元組 (64 hex 字元)")
	}
	internalPrivateKey = ed25519.NewKeyFromSeed(seedBytes)
	return nil
}

func SignPayload(data []byte) (sigHex string, pubHex string) {
	// PureEd25519 直接簽署原始位元組資料，產出 64 bytes 簽名
	signature := ed25519.Sign(internalPrivateKey, data)
	
	// ed25519.SignatureSize 常數恰為 64
	fmt.Printf("Signature Size: %d bytes\n", len(signature))

	pubKey := internalPrivateKey.Public().(ed25519.PublicKey)
	return hex.EncodeToString(signature), hex.EncodeToString(pubKey)
}

func VerifyPayload(data []byte, signatureHex string, publicKeyHex string) bool {
	sig, _ := hex.DecodeString(signatureHex)
	pub, _ := hex.DecodeString(publicKeyHex)
	
	return ed25519.Verify(pub, data, sig)
}
